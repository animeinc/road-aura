    <script>

        let isRecording = false;

        let dataLog = [];

        let potholeCount = 0;

        let currentGps = { lat: 0, lng: 0, speed: 0 };

        

        // 阈值设定（超过此阈值判定为剧烈颠簸/暗坑）

        const POTHOLE_THRESHOLD = 15.0; // Z轴冲击阈值 (m/s^2)

        const waveHistory = new Array(60).fill(0);


        const canvas = document.getElementById('waveform');

        const ctx = canvas.getContext('2d');

        canvas.width = canvas.clientWidth;

        canvas.height = canvas.clientHeight;


        // 绘制实时波形

        function drawWaveform(newVal) {

            waveHistory.push(newVal);

            waveHistory.shift();


            ctx.clearRect(0, 0, canvas.width, canvas.height);

            ctx.strokeStyle = '#38bdf8';

            ctx.lineWidth = 2;

            ctx.beginPath();


            const step = canvas.width / (waveHistory.length - 1);

            const midY = canvas.height / 2;


            for (let i = 0; i < waveHistory.length; i++) {

                // 将震动值映射到画布高度

                const y = midY - (waveHistory[i] * 3);

                if (i === 0) ctx.moveTo(0, y);

                else ctx.lineTo(i * step, y);

            }

            ctx.stroke();


            // 绘制安全阈值红线

            ctx.strokeStyle = '#ef4444';

            ctx.setLineDash([4, 4]);

            ctx.beginPath();

            ctx.moveTo(0, midY - (POTHOLE_THRESHOLD * 3));

            ctx.lineTo(canvas.width, midY - (POTHOLE_THRESHOLD * 3));

            ctx.stroke();

            ctx.setLineDash([]);

        }


        // 传感器处理核心逻辑

        function handleMotion(event) {

            if (!isRecording) return;


            // 获取线性加速度（若浏览器不支持纯线性，则取带重力加速度）

            const acc = event.acceleration && event.acceleration.z !== null 

                        ? event.acceleration 

                        : event.accelerationIncludingGravity;

            

            const zVal = Math.abs(acc.z || 0);

            document.getElementById('val-z').innerText = zVal.toFixed(2);

            drawWaveform(zVal);


            // 判定是否触发坑洼

            const statusBox = document.getElementById('status-box');

            if (zVal > POTHOLE_THRESHOLD) {

                statusBox.className = 'warning';

                statusBox.innerText = '⚠️ 检测到剧烈路面冲击（疑似暗坑）！';

                potholeCount++;

                document.getElementById('val-count').innerText = `${potholeCount} 处`;

                

                // 记录异常点数据

                dataLog.push({

                    time: new Date().toISOString(),

                    lat: currentGps.lat,

                    lng: currentGps.lng,

                    speed: currentGps.speed,

                    impact_z: zVal.toFixed(2),

                    type: 'Pothole_Alert'

                });

            } else {

                statusBox.className = '';

                statusBox.innerText = '正在平稳巡检路况...';

            }

        }


        // GPS 定位跟踪

        function trackGps() {

            if (navigator.geolocation) {

                navigator.geolocation.watchPosition((pos) => {

                    currentGps.lat = pos.coords.latitude.toFixed(6);

                    currentGps.lng = pos.coords.longitude.toFixed(6);

                    currentGps.speed = pos.coords.speed ? (pos.coords.speed * 3.6).toFixed(1) : 0.0;


                    document.getElementById('val-gps').innerText = `${currentGps.lat}, ${currentGps.lng}`;

                    document.getElementById('val-speed').innerText = currentGps.speed;

                }, (err) => {

                    console.warn("GPS 获取受限或处于室内", err);

                }, { enableHighAccuracy: true });

            }

        }


        // 启动/授权采集

        async function startCollector() {

            // iOS 13+ 特别授权流程

            if (typeof DeviceMotionEvent !== 'undefined' && typeof DeviceMotionEvent.requestPermission === 'function') {

                try {

                    const state = await DeviceMotionEvent.requestPermission();

                    if (state !== 'granted') {

                        alert('需要传感器权限方可采集！');

                        return;

                    }

                } catch (e) {

                    console.error(e);

                }

            }


            isRecording = !isRecording;

            const btn = document.getElementById('btn-start');

            if (isRecording) {

                window.addEventListener('devicemotion', handleMotion);

                trackGps();

                btn.innerText = '停止采集';

                btn.style.background = '#e11d48';

                document.getElementById('btn-export').style.display = 'block';

            } else {

                window.removeEventListener('devicemotion', handleMotion);

                btn.innerText = '继续采集';

                btn.style.background = '#0284c7';

                document.getElementById('status-box').innerText = '采集已暂停';

            }

        }


        // 导出 CSV 文件（用于论文/PPT作图）

        function exportCsv() {

            if (dataLog.length === 0) {

                alert('当前未记录到颠簸数据，骑车压过几个坎后再试！');

                return;

            }

            let csvContent = "data:text/csv;charset=utf-8,Time,Latitude,Longitude,Speed_kmh,Impact_Z,Type\n";

            dataLog.forEach(row => {

                csvContent += `${row.time},${row.lat},${row.lng},${row.speed},${row.impact_z},${row.type}\n`;

            });

            const encodedUri = encodeURI(csvContent);

            const link = document.createElement("a");

            link.setAttribute("href", encodedUri);

            link.setAttribute("download", `road_data_${Date.now()}.csv`);

            document.body.appendChild(link);

            link.click();

            document.body.removeChild(link);

        }


        document.getElementById('btn-start').addEventListener('click', startCollector);

        document.getElementById('btn-export').addEventListener('click', exportCsv);

    </script>
